{{- if contains "LoadBalancer" .Values.caddy.service.type }}

        kubectl get svc --namespace {{ .Release.Namespace }} {{ include "oc8.fullname" . }}-caddy -w
{{- else if contains "NodePort" .Values.caddy.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "oc8.fullname" . }}-caddy)
  export NODE_IP=$(kubectl get nodes -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT/
{{- else }}
  kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "oc8.fullname" . }}-caddy 8080:80
  echo http://127.0.0.1:8080/
{{- end }}
